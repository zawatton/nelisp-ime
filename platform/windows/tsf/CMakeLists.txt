cmake_minimum_required(VERSION 3.16)
project(nelisp-ime-tsf LANGUAGES CXX)
add_library(nelisp-ime-tsf SHARED nelisp-ime-tsf.cpp ../../common/nelisp_ime_client.cpp nelisp-ime-tsf.def)
target_compile_features(nelisp-ime-tsf PRIVATE cxx_std_17)
target_include_directories(nelisp-ime-tsf PRIVATE ../../common)
target_compile_definitions(nelisp-ime-tsf PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(nelisp-ime-tsf PRIVATE /W4 /WX)
else()
  target_compile_options(nelisp-ime-tsf PRIVATE -Wall -Wextra -Werror)
endif()
target_link_libraries(nelisp-ime-tsf PRIVATE ole32 uuid advapi32)
set_target_properties(nelisp-ime-tsf PROPERTIES PREFIX "")
add_executable(nelisp-ime-tsf-smoke smoke.cpp)
target_compile_features(nelisp-ime-tsf-smoke PRIVATE cxx_std_17)
target_link_libraries(nelisp-ime-tsf-smoke PRIVATE ole32 uuid)
if(NOT MSVC)
  target_link_options(nelisp-ime-tsf PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(nelisp-ime-tsf-smoke PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(nelisp-ime-tsf-smoke PRIVATE -municode)
endif()
