cmake_minimum_required(VERSION 3.16)
project(fcitx5-nelisp-ime VERSION 0.1.0 LANGUAGES CXX)
find_package(Fcitx5Core REQUIRED)
find_package(Threads REQUIRED)
include("${Fcitx5Utils_DIR}/Fcitx5CompilerSettings.cmake")
add_library(nelisp-ime SHARED nelisp-ime.cpp ../../common/nelisp_ime_client.cpp)
target_compile_features(nelisp-ime PRIVATE cxx_std_17)
target_include_directories(nelisp-ime PRIVATE ../../common)
target_link_libraries(nelisp-ime PRIVATE Fcitx5::Core Threads::Threads)
if(MSVC)
  target_compile_options(nelisp-ime PRIVATE /W4 /WX)
else()
  target_compile_options(nelisp-ime PRIVATE -Wall -Wextra -Werror)
endif()
set_target_properties(nelisp-ime PROPERTIES PREFIX "")
configure_file(nelisp-ime-addon.conf.in nelisp-ime-addon.conf)
configure_file(nelisp-ime.conf.in nelisp-ime.conf)
install(TARGETS nelisp-ime DESTINATION "${FCITX_INSTALL_LIBDIR}/fcitx5")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nelisp-ime-addon.conf" RENAME nelisp-ime.conf DESTINATION "${FCITX_INSTALL_PKGDATADIR}/addon")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nelisp-ime.conf" DESTINATION "${FCITX_INSTALL_PKGDATADIR}/inputmethod")

option(NELISP_IME_BUILD_TESTS "Build native Fcitx5 recovery tests" OFF)
if(NELISP_IME_BUILD_TESTS)
  include(CTest)
  add_executable(nelisp-ime-fcitx5-recovery-runtime recovery_fixture_runtime.cpp)
  add_executable(nelisp-ime-fcitx5-recovery-test recovery_fixture.cpp ../../common/nelisp_ime_client.cpp)
  target_compile_features(nelisp-ime-fcitx5-recovery-runtime PRIVATE cxx_std_17)
  target_compile_features(nelisp-ime-fcitx5-recovery-test PRIVATE cxx_std_17)
  target_include_directories(nelisp-ime-fcitx5-recovery-test PRIVATE ../../common)
  target_link_libraries(nelisp-ime-fcitx5-recovery-test PRIVATE Fcitx5::Core Threads::Threads)
  if(MSVC)
    target_compile_options(nelisp-ime-fcitx5-recovery-test PRIVATE /W4 /WX)
    target_compile_options(nelisp-ime-fcitx5-recovery-runtime PRIVATE /W4 /WX)
  else()
    target_compile_options(nelisp-ime-fcitx5-recovery-test PRIVATE -Wall -Wextra -Werror)
    target_compile_options(nelisp-ime-fcitx5-recovery-runtime PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME nelisp-ime-fcitx5-recovery
    COMMAND nelisp-ime-fcitx5-recovery-test $<TARGET_FILE:nelisp-ime-fcitx5-recovery-runtime>)
  set_tests_properties(nelisp-ime-fcitx5-recovery PROPERTIES TIMEOUT 20)
endif()
